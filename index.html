#include #include #include "HX711.h" #include #define DOUT D6 #define SCK D5 // LED and buzzer #define LED_PIN D7 #define BUZZER_PIN D2 const char* ssid = "Manavi_10"; const char* password = "manavi1002"; ESP8266WebServer server(80); HX711 scale; // ===================================================== // CALIBRATION // ===================================================== float calibration_factor = 420.0; // ===================================================== // INVENTRA SETTINGS // ===================================================== const int MAX_PRODUCTS = 4; const int PRODUCT_WEIGHT = 250; const int MAX_CAPACITY = 1000; // Keep these settings because your current load-cell // detection is already detecting placement. const float DETECTION_THRESHOLD = 0.30; const float MOVEMENT_THRESHOLD = 0.08; const unsigned long SETTLE_TIME = 1200; // ===================================================== // PRODUCT AUTHORIZATION // ===================================================== // Products scanned on the phone but not yet placed String authorizedProducts[MAX_PRODUCTS]; int authorizedCount = 0; // Products actually authenticated and placed String trolleyProducts[MAX_PRODUCTS]; int trolleyCount = 0; // Extra product warning bool extraProductDetected = false; // ===================================================== // LOAD CELL VARIABLES // ===================================================== float stableWeight = 0; float candidateWeight = 0; bool waitingForStable = false; unsigned long changeStartTime = 0; // ===================================================== // CORS // ===================================================== void allowCORS() { server.sendHeader( "Access-Control-Allow-Origin", "*" ); } // ===================================================== // LED + BUZZER // ===================================================== void extraAlertOn() { digitalWrite(LED_PIN, HIGH); digitalWrite(BUZZER_PIN, HIGH); } void extraAlertOff() { digitalWrite(LED_PIN, LOW); digitalWrite(BUZZER_PIN, LOW); } // ===================================================== // PRINT STATUS // ===================================================== void printStatus() { Serial.println(); Serial.println("================================"); Serial.println(" INVENTRA STATUS"); Serial.println("================================"); Serial.print("Scanned Products: "); Serial.println( authorizedCount + trolleyCount ); Serial.print("Products in Trolley: "); Serial.println(trolleyCount); Serial.print("Virtual Trolley Weight: "); Serial.print( trolleyCount * PRODUCT_WEIGHT ); Serial.println(" g"); Serial.println(); if (authorizedCount > 0) { Serial.println( "Waiting for placement:" ); for (int i = 0; i < authorizedCount; i++) { Serial.print(" "); Serial.println( authorizedProducts[i] ); } } if (trolleyCount > 0) { Serial.println(); Serial.println( "Authenticated Products:" ); for (int i = 0; i < trolleyCount; i++) { Serial.print(" "); Serial.println( trolleyProducts[i] ); } } Serial.println("================================"); } // ===================================================== // JSON ARRAY // ===================================================== String makeProductArray( String products[], int count ) { String result = "["; for (int i = 0; i < count; i++) { if (i > 0) { result += ","; } result += "\""; result += products[i]; result += "\""; } result += "]"; return result; } // ===================================================== // SEND DATA TO WEBSITE // ===================================================== void sendData() { int scannedCount = authorizedCount + trolleyCount; int logicalWeight = trolleyCount * PRODUCT_WEIGHT; String json = "{"; json += "\"connected\":true,"; json += "\"scannedCount\":"; json += String(scannedCount); json += ","; json += "\"trolleyCount\":"; json += String(trolleyCount); json += ","; json += "\"logicalWeight\":"; json += String(logicalWeight); json += ","; json += "\"capacity\":"; json += String(MAX_CAPACITY); json += ","; json += "\"extra\":"; json += ( extraProductDetected ? "true" : "false" ); json += ","; json += "\"capacityExceeded\":false,"; json += "\"authorizedProducts\":"; json += makeProductArray( authorizedProducts, authorizedCount ); json += ","; json += "\"trolleyProducts\":"; json += makeProductArray( trolleyProducts, trolleyCount ); json += "}"; allowCORS(); server.send( 200, "application/json", json ); } // ===================================================== // AUTHORIZE SCANNED PRODUCT // ===================================================== void authorizeProduct() { if (!server.hasArg("name")) { allowCORS(); server.send( 400, "text/plain", "Product name missing" ); return; } String productName = server.arg("name"); productName.trim(); if (productName.length() == 0) { allowCORS(); server.send( 400, "text/plain", "Invalid product" ); return; } int totalProducts = authorizedCount + trolleyCount; if (totalProducts >= MAX_PRODUCTS) { allowCORS(); server.send( 400, "text/plain", "Trolley capacity reached" ); return; } authorizedProducts[ authorizedCount ] = productName; authorizedCount++; Serial.println(); Serial.println("================================"); Serial.println("SCANNED PRODUCT"); Serial.print("Product: "); Serial.println(productName); Serial.print( "Authorized Products: " ); Serial.println(authorizedCount); Serial.println("================================"); allowCORS(); server.send( 200, "text/plain", "Product authorized" ); } // ===================================================== // RESET TROLLEY // ===================================================== void resetTrolley() { authorizedCount = 0; trolleyCount = 0; extraProductDetected = false; extraAlertOff(); for (int i = 0; i < MAX_PRODUCTS; i++) { authorizedProducts[i] = ""; trolleyProducts[i] = ""; } Serial.println(); Serial.println( "================================" ); Serial.println( "INVENTRA TROLLEY RESET" ); Serial.println( "Waiting for product scan..." ); Serial.println( "================================" ); } // ===================================================== // RESET HTTP REQUEST // ===================================================== void handleReset() { resetTrolley(); allowCORS(); server.send( 200, "text/plain", "Trolley reset" ); } // ===================================================== // REMOVE FIRST AUTHORIZED PRODUCT // ===================================================== String removeFirstAuthorizedProduct() { if (authorizedCount <= 0) { return ""; } String product = authorizedProducts[0]; for ( int i = 0; i < authorizedCount - 1; i++ ) { authorizedProducts[i] = authorizedProducts[i + 1]; } authorizedProducts[ authorizedCount - 1 ] = ""; authorizedCount--; return product; } // ===================================================== // HANDLE PRODUCT PLACEMENT // ===================================================== void handleProductPlacement() { Serial.println(); Serial.println( "================================" ); // --------------------------------------------------- // AUTHORIZED PRODUCT // --------------------------------------------------- if (authorizedCount > 0) { if (trolleyCount < MAX_PRODUCTS) { String product = removeFirstAuthorizedProduct(); trolleyProducts[ trolleyCount ] = product; trolleyCount++; extraProductDetected = false; extraAlertOff(); Serial.println( "PRODUCT PLACED" ); Serial.print( "Authenticated: " ); Serial.println(product); Serial.print( "Product Count: " ); Serial.println(trolleyCount); Serial.print( "Trolley Weight: " ); Serial.print( trolleyCount * PRODUCT_WEIGHT ); Serial.println(" g"); } } // --------------------------------------------------- // EXTRA PRODUCT // --------------------------------------------------- else { extraProductDetected = true; extraAlertOn(); Serial.println( "!!! EXTRA PRODUCT DETECTED !!!" ); Serial.println( "No scanned product authorization available." ); Serial.print( "Products in Trolley: " ); Serial.println(trolleyCount); Serial.println( "BUZZER + LED ON" ); } Serial.println( "================================" ); } // ===================================================== // HANDLE PRODUCT REMOVAL // ===================================================== void handleProductRemoval() { // If an extra product was removed, // simply clear the warning. if (extraProductDetected) { extraProductDetected = false; extraAlertOff(); Serial.println(); Serial.println( "EXTRA PRODUCT REMOVED" ); Serial.println( "Trolley status normal." ); return; } // Remove last authenticated product if (trolleyCount > 0) { Serial.println(); Serial.println( "PRODUCT REMOVED" ); Serial.print( "Removed: " ); Serial.println( trolleyProducts[ trolleyCount - 1 ] ); trolleyProducts[ trolleyCount - 1 ] = ""; trolleyCount--; Serial.print( "Products in Trolley: " ); Serial.println(trolleyCount); Serial.print( "Trolley Weight: " ); Serial.print( trolleyCount * PRODUCT_WEIGHT ); Serial.println(" g"); } } // ===================================================== // SETUP // ===================================================== void setup() { Serial.begin(115200); delay(500); Serial.println(); Serial.println( "================================" ); Serial.println( " INVENTRA SMART TROLLEY" ); Serial.println( "================================" ); // LED pinMode( LED_PIN, OUTPUT ); // Buzzer pinMode( BUZZER_PIN, OUTPUT ); extraAlertOff(); // --------------------------------------------------- // HX711 // --------------------------------------------------- scale.begin( DOUT, SCK ); scale.set_scale( calibration_factor ); Serial.println( "Keep trolley empty for zeroing." ); delay(2000); scale.tare(); delay(500); stableWeight = scale.get_units(10); candidateWeight = stableWeight; // --------------------------------------------------- // WIFI // --------------------------------------------------- WiFi.mode(WIFI_STA); WiFi.begin( ssid, password ); Serial.print( "Connecting to WiFi" ); while ( WiFi.status() != WL_CONNECTED ) { delay(500); Serial.print("."); } Serial.println(); Serial.println( "WiFi connected!" ); Serial.print( "ESP8266 IP: " ); Serial.println( WiFi.localIP() ); // --------------------------------------------------- // SERVER // --------------------------------------------------- server.on( "/", []() { allowCORS(); server.send( 200, "text/plain", "INVENTRA ESP8266 SERVER OK" ); } ); server.on( "/data", sendData ); server.on( "/authorize", HTTP_GET, authorizeProduct ); server.on( "/reset", HTTP_GET, handleReset ); server.begin(); Serial.println( "Server started." ); Serial.println(); Serial.println( "Maximum Capacity: 1000 g" ); Serial.println( "Weight per Authenticated Product: 250 g" ); Serial.println(); Serial.println( "Waiting for product scan..." ); Serial.println(); printStatus(); } // ===================================================== // LOOP // ===================================================== void loop() { server.handleClient(); if (!scale.is_ready()) { delay(50); return; } float currentWeight = scale.get_units(5); // --------------------------------------------------- // WAIT FOR A CHANGE // --------------------------------------------------- if (!waitingForStable) { if ( fabs( currentWeight - stableWeight ) >= DETECTION_THRESHOLD ) { candidateWeight = currentWeight; changeStartTime = millis(); waitingForStable = true; } } // --------------------------------------------------- // WAIT UNTIL STABLE // --------------------------------------------------- else { float movement = fabs( currentWeight - candidateWeight ); candidateWeight = currentWeight; if ( movement > MOVEMENT_THRESHOLD ) { changeStartTime = millis(); } if ( millis() - changeStartTime >= SETTLE_TIME ) { float difference = currentWeight - stableWeight; // PRODUCT PLACED if ( difference >= DETECTION_THRESHOLD ) { handleProductPlacement(); } // PRODUCT REMOVED else if ( difference <= -DETECTION_THRESHOLD ) { handleProductRemoval(); } stableWeight = currentWeight; waitingForStable = false; printStatus(); } } delay(100); }

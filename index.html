server.sendHeader( "Access-Control-Allow-Headers", "*" ); } // ===================================================== // ALERT OFF // ===================================================== void alertOff() { digitalWrite( LED_PIN, LOW ); digitalWrite( BUZZER_PIN, LOW ); } // ===================================================== // BUZZER // ===================================================== void beep() { digitalWrite( BUZZER_PIN, HIGH ); beep(); } // ===================================================== // PRODUCT ARRAY → JSON // ===================================================== String makeProductArray( String products[], int count ) { String result = "["; for ( int i = 0; i < trolleyCount; i++ ) { Serial.print(" "); Serial.println( trolleyProducts[i] ); } } if ( extraProductDetected ) { Serial.println(); Serial.println( "⚠ EXTRA PRODUCT DETECTED" ); } Serial.println( "================================" ); } // ===================================================== // SEND DATA TO WEBSITE // ===================================================== void sendData() { int totalScanned = authorizedCount + trolleyCount; int logicalWeight = trolleyCount * PRODUCT_WEIGHT; String json = "{"; json += "\"connected\":true,"; json += "\"scannedCount\":"; json += String(totalScanned); json += ","; json += "\"trolleyCount\":"; json += makeProductArray( trolleyProducts, trolleyCount ); json += "}"; allowCORS(); server.send( 200, "application/json", json ); } // ===================================================== // INITIAL BASELINE // ===================================================== void establishInitialBaseline() { if ( !scale.is_ready() ) { return; } float total = 0; const int samples = 10; for ( int i = 0; i < samples; i++ ) { total += scale.get_units(1); delay(20); } baselineWeight = total / samples; candidateWeight = baselineWeight; checkingChange = false; } // ===================================================== // AUTHORIZE PRODUCT // ===================================================== void authorizeProduct() { if ( !server.hasArg("name") ) { allowCORS(); server.send( 400, "text/plain", "Product name missing" ); return; } String productName = server.arg("name"); productName.trim(); if ( productName.length() == 0 ) { allowCORS(); server.send( 400, "text/plain", "Invalid product" ); return; } // =================================================== // CAPACITY CHECK // =================================================== int totalProducts = authorizedCount + trolleyCount; if ( totalProducts >= MAX_PRODUCTS ) { allowCORS(); server.send( 400, "text/plain", "Trolley capacity reached" ); return; } // =================================================== // IMPORTANT // // DO NOT: // - tare // - update baseline // - reset trolley count // - reset trolley products // // Scanning ONLY authorizes a product. // =================================================== extraProductDetected = false; server.send( 200, "text/plain", "Product authorized" ); } // ===================================================== // REMOVE FIRST AUTHORIZED PRODUCT // ===================================================== String removeFirstAuthorizedProduct() { if ( authorizedCount <= 0 ) { return ""; Serial.println( "TROLLEY STATUS NORMAL" ); Serial.println( "================================" ); } // ===================================================== // LOAD CELL PROCESSING // ===================================================== void processLoadCell() { if ( !scale.is_ready() ) { return; } if ( millis() - lastSensorRead < SENSOR_INTERVAL ) { return; } lastSensorRead = millis(); // --------------------------------------------------- // TAKE MULTIPLE READINGS // --------------------------------------------------- float total = 0; const int READINGS = 3; for ( int i = 0; i < READINGS; i++ ) { total += scale.get_units(1); } float currentWeight = total / READINGS; float difference = currentWeight - baselineWeight; // =================================================== // SCAN SETTLING PERIOD // =================================================== // // IMPORTANT: // We DO NOT change baseline here. // // This prevents a scan from becoming a fake // "product placement". // =================================================== if ( waitingForPlacement && millis() - authorizationTime < SCAN_SETTLE_TIME ) { checkingChange = false; } // =================================================== // CONFIRM THAT WEIGHT HAS SETTLED // =================================================== float movement = fabs( currentWeight - candidateWeight ); } float finalDifference = currentWeight - baselineWeight; // =================================================== // CONFIRMED PRODUCT ADDITION // =================================================== if ( finalDifference >= DETECTION_THRESHOLD ) { // ------------------------------------------------- // AUTHORIZED PRODUCT // ------------------------------------------------- if ( authorizedCount > 0 ) { registerAuthorizedPlacement(); } // ------------------------------------------------- // UNAUTHORIZED PRODUCT // ------------------------------------------------- else { registerExtraProduct(); } // ------------------------------------------------- // VERY IMPORTANT // // Only AFTER a confirmed physical change do we // update the baseline. // ------------------------------------------------- baselineWeight = currentWeight; lastEventTime = millis(); checkingChange = false; candidateWeight = currentWeight; printStatus(); return; } // =================================================== // CONFIRMED REMOVAL // =================================================== if ( finalDifference <= -DETECTION_THRESHOLD ) { // ------------------------------------------------- // If an extra product was detected: // // DO NOT remove any authorized trolley product. // Just clear the extra warning. // ------------------------------------------------- if ( extraRemovalPending ) { clearExtraProduct(); } // ------------------------------------------------- // Normal authorized product removal // // Only do this when there is NO extra warning. // ------------------------------------------------- else { if ( trolleyCount > 0 ) { Serial.println(); Serial.println( "AUTHORIZED PRODUCT REMOVED" ); Serial.print( "Removed: " ); Serial.println( trolleyProducts[ trolleyCount - 1 ] ); trolleyProducts[ trolleyCount - 1 ] = ""; trolleyCount--; Serial.print( "Products in Trolley: " ); Serial.println( trolleyCount ); Serial.print( "Virtual Weight: " ); Serial.print( trolleyCount * PRODUCT_WEIGHT ); Serial.println( " g" ); } } // New physical state is now baseline. baselineWeight = currentWeight; lastEventTime = millis(); checkingChange = false; candidateWeight = currentWeight; printStatus(); return; } // =================================================== // RESET DETECTION // =================================================== checkingChange = false; candidateWeight = currentWeight; } // ===================================================== // RESET TROLLEY // ===================================================== void resetTrolley() { authorizedCount = 0; trolleyCount = 0; extraProductDetected = false; extraRemovalPending = false; waitingForPlacement = false; checkingChange = false; alertOff(); for ( int i = 0; i < MAX_PRODUCTS; i++ ) { authorizedProducts[i] = "";
